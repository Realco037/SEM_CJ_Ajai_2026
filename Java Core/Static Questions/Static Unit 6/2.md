class Task implements Comparable<Task> {
    String name;
    Date deadline;
    int priority;
    private static final SimpleDateFormat dateFormat = new SimpleDateFormat("yyyy-MM-dd");
    public Task(String name, String deadline, int priority) {
        this.name = name;
        this.deadline = dateFormat.parse(deadline, new ParsePosition(0));
        this.priority = priority;
    }

    public int compareTo(Task other) {
        int dateComparison = this.deadline.compareTo(other.deadline);
        if (dateComparison != 0) {
            return dateComparison;
        }
        return Integer.compare(this.priority, other.priority);
    }

    public String toString() {
        return name + " " + dateFormat.format(deadline) + " " + priority;
    }
}
